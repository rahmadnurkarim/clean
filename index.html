Jalankan skrip pemindai di Console browser pada tab instagram.com untuk mengumpulkan daftar Followers dan Following lengkap dengan ID akun numerik (diperlukan untuk fitur Unfollow Console Anti-Block), atau impor arsip resmi Instagram.

KODE SCRAPER LANGSUNG (JALANKAN DI TAB INSTAGRAM) 
/* =========================================================================
 * INSTACLEAN DUAL SCANNER — FOLLOWERS & FOLLOWING (SAFE MODE)
 * Jalankan di Console (F12) pada tab instagram.com saat sudah login.
 * ========================================================================= */
(async () => {
    const log = (msg, color = "#06b6d4") =>
        console.log(`%c[InstaClean] ${msg}`, `color:${color};font-weight:bold`);
    const sleep = ms => new Promise(resolve => setTimeout(resolve, ms));

    const getCookie = name => {
        const escaped = name.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
        const match = document.cookie.match(new RegExp(`(?:^|; )${escaped}=([^;]*)`));
        return match ? decodeURIComponent(match[1]) : null;
    };

    const userId = getCookie("ds_user_id");
    const csrf = getCookie("csrftoken");
    if (!userId) {
        log("Cookie ds_user_id tidak ditemukan. Pastikan sudah login di instagram.com.", "#ef4444");
        return;
    }

    const headers = {
        "Accept": "application/json, text/plain, */*",
        "X-IG-App-ID": "936619743392459",
        "X-Requested-With": "XMLHttpRequest"
    };
    if (csrf) headers["X-CSRFToken"] = csrf;

    async function requestPage(kind, cursor) {
        const params = new URLSearchParams({ count: "100" });
        if (cursor) params.set("max_id", cursor);
        const url = `https://www.instagram.com/api/v1/friendships/${userId}/${kind}/?${params}`;

        const response = await fetch(url, {
            method: "GET", credentials: "include", headers, cache: "no-store"
        });
        const contentType = response.headers.get("content-type") || "";
        const finalUrl = response.url || url;
        const body = await response.text();
        const looksLikeHtml = /text\/html/i.test(contentType) || /^\s*(<!doctype|<html)/i.test(body);

        if (response.redirected || looksLikeHtml || /\/(accounts\/login|challenge|checkpoint)/i.test(finalUrl)) {
            throw new Error(`Instagram meminta verifikasi/login (HTTP ${response.status}).`);
        }
        if (response.status === 429) throw new Error("HTTP 429: Terkena batas sementara (Rate Limit). Hentikan dan coba lagi nanti.");
        if (!response.ok) throw new Error(`HTTP ${response.status}: Permintaan gagal.`);

        const json = JSON.parse(body);
        if (json.status && json.status !== "ok") throw new Error(json.message || `Status: ${json.status}`);
        if (!Array.isArray(json.users)) throw new Error("Format respons tidak berisi daftar users.");
        return json;
    }

    async function scan(kind, label) {
        const collected = new Map();
        const seenCursors = new Set();
        let cursor = null, page = 0;

        while (true) {
            page++;
            if (page > 500) break;
            const data = await requestPage(kind, cursor);
            const users = Array.isArray(data.users) ? data.users : [];

            for (const u of users) {
                const rawId = u && (u.pk || u.id);
                if (!u || !rawId || !u.username) continue;
                const id = String(rawId);
                collected.set(id, {
                    id,
                    username: u.username,
                    full_name: u.full_name || "",
                    profile_pic_url: u.profile_pic_url || "",
                    is_private: Boolean(u.is_private),
                    is_verified: Boolean(u.is_verified)
                });
            }

            log(`${label}: halaman ${page} selesai (total terkumpul: ${collected.size} akun)`);
            const nextCursor = data.next_max_id || null;
            if (!nextCursor || seenCursors.has(String(nextCursor)) || users.length === 0) break;

            seenCursors.add(String(nextCursor));
            cursor = String(nextCursor);
            // Jeda acak 2.5 - 4.5 detik antar halaman agar aman dari blokir
            await sleep(2500 + Math.floor(Math.random() * 2000));
        }
        return [...collected.values()];
    }

    try {
        log("Memulai pemindaian Following...");
        const followingList = await scan("following", "Following");
        await sleep(3000);
        log("Memulai pemindaian Followers...");
        const followersList = await scan("followers", "Followers");

        const followerIds = new Set(followersList.map(u => u.id));
        const allUsers = new Map();
        for (const u of followingList) {
            allUsers.set(u.id, { ...u, is_following: true, is_follower: followerIds.has(u.id) });
        }
        for (const u of followersList) {
            if (allUsers.has(u.id)) allUsers.get(u.id).is_follower = true;
            else allUsers.set(u.id, { ...u, is_following: false, is_follower: true });
        }

        const accounts = [...allUsers.values()];
        const summary = {
            account_id: userId,
            scanned_at: new Date().toISOString(),
            following_count: followingList.length,
            followers_count: followersList.length,
            unique_relations: accounts.length,
            accounts
        };

        const blob = new Blob([JSON.stringify(summary, null, 2)], { type: "application/json;charset=utf-8" });
        const link = document.createElement("a");
        link.href = URL.createObjectURL(blob);
        link.download = `instaclean_relations_${userId}.json`;
        document.body.appendChild(link);
        link.click();
        link.remove();

        log(`SELESAI! ${accounts.length} akun disimpan ke instaclean_relations_${userId}.json`, "#10b981");
    } catch (error) {
        log(`PEMINDAIAN DIHENTIKAN: ${error.message}`, "#ef4444");
    }
})();

Cara pakai: Buka Instagram di browser dan login. Tekan F12 → tab Console (ketik allow pasting jika diminta Chrome/Edge), tempel kode di atas, lalu tekan Enter. Setelah file .json terunduh otomatis, klik tombol Impor JSON di pojok kanan atas aplikasi ini.
