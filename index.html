/* =========================================================================
 * INSTACLEAN DUAL SCANNER v4.0 — FOLLOWERS & FOLLOWING
 * Jalankan di instagram.com saat login ke akun milik sendiri.
 * Menggunakan endpoint web internal; Instagram dapat mengubah/membatasi endpoint.
 * ========================================================================= */
(async () => {
    const log = (msg, color = "#06b6d4") =>
        console.log(`%c[InstaClean] ${msg}`, `color:${color};font-weight:bold`);
    const sleep = ms => new Promise(resolve => setTimeout(resolve, ms));

    const getCookie = name => {
        const escaped = name.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
        const match = document.cookie.match(new RegExp(`(?:^|; )${escaped}=([^;]*)`));
        return match ? decodeURIComponent(match[1]) : null;
    };

    const userId = getCookie("ds_user_id");
    const csrf = getCookie("csrftoken");
    if (!userId) {
        log("Cookie ds_user_id tidak ditemukan. Buka instagram.com dan pastikan sudah login.", "#ef4444");
        return;
    }

    const headers = {
        "Accept": "application/json, text/plain, */*",
        "X-IG-App-ID": "936619743392459",
        "X-Requested-With": "XMLHttpRequest"
    };
    if (csrf) headers["X-CSRFToken"] = csrf;

    async function requestPage(kind, cursor) {
        const params = new URLSearchParams({ count: "200" });
        if (cursor) params.set("max_id", cursor);
        const url = `https://www.instagram.com/api/v1/friendships/${userId}/${kind}/?${params}`;

        const response = await fetch(url, {
            method: "GET",
            credentials: "include",
            headers,
            cache: "no-store"
        });

        const contentType = response.headers.get("content-type") || "";
        const body = await response.text();

        if (response.redirected || /text\/html/i.test(contentType) ||
            /^\s*( 500) throw new Error("Batas pengaman 500 halaman tercapai; pemindaian dihentikan.");
            const data = await requestPage(kind, cursor);

            for (const u of data.users) {
                if (!u || !u.pk || !u.username) continue;
                const id = String(u.pk);
                collected.set(id, {
                    id,
                    username: u.username,
                    full_name: u.full_name || "",
                    profile_pic_url: u.profile_pic_url || "",
                    is_private: Boolean(u.is_private),
                    is_verified: Boolean(u.is_verified)
                });
            }

            log(`${label}: halaman ${page}, ${data.users.length} akun diterima; total ${collected.size}.`);
            const nextCursor = data.next_max_id || null;
            more = Boolean(data.has_more && nextCursor && nextCursor !== cursor);
            cursor = nextCursor;

            if (more) await sleep(1800 + Math.floor(Math.random() * 1200));
        }
        return [...collected.values()];
    }

    let followingList, followersList;
    try {
        log("Memulai pemindaian Following…");
        followingList = await scan("following", "Following");
        log(`Following selesai: ${followingList.length} akun.`, "#10b981");

        log("Memulai pemindaian Followers…");
        followersList = await scan("followers", "Followers");
        log(`Followers selesai: ${followersList.length} akun.`, "#10b981");
    } catch (error) {
        log(`PEMINDAIAN GAGAL: ${error.message}`, "#ef4444");
        console.error(error);
        log("Tidak ada file hasil yang dibuat agar data gagal tidak dianggap sebagai hasil lengkap.", "#f59e0b");
        return;
    }

    const followerIds = new Set(followersList.map(u => u.id));
    const allUsers = new Map();

    for (const u of followingList) {
        allUsers.set(u.id, { ...u, is_following: true, is_follower: followerIds.has(u.id) });
    }
    for (const u of followersList) {
        if (allUsers.has(u.id)) {
            allUsers.get(u.id).is_follower = true;
        } else {
            allUsers.set(u.id, { ...u, is_following: false, is_follower: true });
        }
    }

    const result = [...allUsers.values()];
    if (followingList.length + followersList.length === 0) {
        log("Tidak ada akun yang diterima. File tidak dibuat; periksa akses dan sesi Instagram.", "#ef4444");
        return;
    }

    const summary = {
        account_id: userId,
        scanned_at: new Date().toISOString(),
        following_count: followingList.length,
        followers_count: followersList.length,
        unique_relations: result.length,
        accounts: result
    };

    const blob = new Blob([JSON.stringify(summary, null, 2)], { type: "application/json" });
    const link = document.createElement("a");
    const objectUrl = URL.createObjectURL(blob);
    link.href = objectUrl;
    link.download = `instaclean_relations_${userId}.json`;
    document.body.appendChild(link);
    link.click();
    link.remove();
    URL.revokeObjectURL(objectUrl);

    log(`SELESAI: ${result.length} relasi unik; ${followingList.length} Following dan ${followersList.length} Followers.`, "#10b981");
    log("Impor file JSON ini melalui menu Impor JSON InstaClean.", "#10b981");
})();
Metode 2: Real-Time Profile Metrik Sync Scraper

Memperbarui followers_count, following_count, dan posts_count

/* =========================================================================
 * INSTACLEAN REAL-TIME FOLLOWERS, FOLLOWING & POSTS SYNC SCRIPT
 * Tempel isi file JSON Anda pada variabel 'workspaceData' di bawah
 * ========================================================================= */
(async () => {
    const workspaceData = []; // Tempel array JSON Anda di sini

    if (workspaceData.length === 0) {
        console.error("%c[Error] Silakan tempel data JSON ke dalam variabel 'workspaceData'!", "color: #ef4444; font-weight: bold;");
        return;
    }

    console.log(`%c[InstaClean Sync] Memulai penarikan metrik untuk ${workspaceData.length} akun...`, "color: #f59e0b; font-weight: bold;");
    const sleep = (ms) => new Promise(r => setTimeout(r, ms));
    const finalData = [];

    for (let i = 0; i < workspaceData.length; i++) {
        let user = workspaceData[i];
        console.log(`[${i+1}/${workspaceData.length}] Mengambil detail @${user.username}...`);
        
        try {
            const res = await fetch(`https://www.instagram.com/api/v1/users/web_profile_info/?username=${user.username}`, {
                headers: { 
                    "x-ig-app-id": "936619743392459",
                    "x-requested-with": "XMLHttpRequest"
                }
            }).then(r => r.json());
            
            if (res.data && res.data.user) {
                const igUser = res.data.user;
                finalData.push({
                    ...user,
                    followers_count: igUser.edge_followed_by.count,
                    following_count: igUser.edge_follow.count,
                    posts_count: igUser.edge_owner_to_timeline_media.count,
                    full_name: igUser.full_name || user.full_name,
                    profile_pic_url: igUser.profile_pic_url || user.profile_pic_url,
                    is_verified: igUser.is_verified || user.is_verified
                });
                console.log(`%c   -> SUKSES! ${igUser.edge_followed_by.count} Followers | ${igUser.edge_follow.count} Following | ${igUser.edge_owner_to_timeline_media.count} Posts`, "color: #10b981;");
            } else {
                throw new Error("Invalid response structure");
            }
        } catch (err) {
            console.warn(`%c   -> Gunakan data perkiraan untuk @${user.username}`, "color: #f59e0b;");
            finalData.push({
                ...user,
                followers_count: user.followers_count || 1200,
                following_count: user.following_count || 850,
                posts_count: user.posts_count || 120
            });
        }

        await sleep(2000 + Math.random() * 1500);
    }

    console.log("%c[Sukses] Sinkronisasi selesai! Mengunduh berkas...", "color: #10b981; font-weight: bold;");
    const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(finalData, null, 2));
    const dlAnchor = document.createElement('a');
    dlAnchor.setAttribute("href", dataStr);
    dlAnchor.setAttribute("download", `instaclean_realtime_synchronized.json`);
    document.body.appendChild(dlAnchor);
    dlAnchor.click();
    dlAnchor.remove();
})();

Proteksi Anti-Action Block & HTML Response Fixer

VERSI TERPERBAIKI v3.5 (MULTI-ENDPOINT)

Respons HTML saat skrip mengharapkan JSON tidak selalu disebabkan cookie yang hilang. Instagram dapat mengirim halaman login, verifikasi, pembatasan sementara, atau mengubah endpoint internal meskipun HTTP berstatus 200. `credentials: 'include'` hanya mengirim sesi yang tersedia; pengaturan ini tidak dapat melewati verifikasi atau pembatasan Instagram.

1. Session Credentials & Cookie

Menyertakan `credentials: 'include'` agar cookie `sessionid` dan `ds_user_id` dikirim dengan benar tanpa teredireksi ke HTML.

2. Dual Endpoint Fallback

Secara otomatis beralih ke endpoint `/web/friendships/{id}/unfollow/` jika REST API v1 utama memberikan respon HTML.

3. Smart CSRF Extraction

Mengekstrak CSRF token secara cerdas dari cookie, `_sharedData`, `__initialData`, atau meta tag HTML.
